import http from 'k6/http';

import ScenarioUtils from '../utils/scenarioUtils.js';
import Utils from '../utils/utils.js';

// A copy of the health scenario whose check expects 200, while GET /api/health answers 204: every check
// fails, the latency stays within its budget, so only the success-rate threshold of G22 can fail the run.
const scenarioName = 'planted_failing_check';

const utils = new Utils();
const scenarioUtils = new ScenarioUtils(utils, scenarioName);

export const options = scenarioUtils.getOptions();

export default function plantedFailingCheck() {
  const response = http.get(`${utils.getBaseHttpUrl()}/health`);
  utils.checkResponse(response, 'is status 200', res => res.status === 200);
}
